# Packages for this setup. All from the official Manjaro/Arch repos — no AUR
# required. Install with:
#
#     sudo pacman -S --needed - < packages.txt
#
# (install.sh does this for you, skipping anything already present.)
# Lines starting with # and blank lines are stripped by install.sh.

# --- window manager & desktop ----------------------------------------------
i3-wm
polybar
rofi
dunst
picom
autotiling                 # automatic split direction for i3
xorg-xrandr
xorg-xsetroot
xorg-xinput

xorg-xrdb
xorg-xset

tmux
zsh

zsh-syntax-highlighting

zsh-theme-powerlevel10k

noto-fonts-cjk
